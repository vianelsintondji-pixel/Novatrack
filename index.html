📋 Guide de configuration

Suis ces étapes pour connecter NovaTrack à Google Sheets :

  1. Va sur sheets.google.com et crée un nouveau tableur. Nomme-le NovaTrack.
  2. Dans le tableur, clique sur Extensions → Apps Script.
  3. Supprime le code existant et colle le code Apps Script fourni ci-dessous.
  4. Clique sur Déployer → Nouveau déploiement. Type : Application Web. Accès : Tout le monde.
  5. Autorise les permissions puis copie l'URL de déploiement.
  6. Colle cette URL dans le champ ci-dessus et clique sur Enregistrer.

📄 Code Apps Script à coller

Copie tout ce code dans l'éditeur Apps Script :

const SHEET_NAME = 'Livraisons';

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME) ||
    SpreadsheetApp.getActiveSpreadsheet().insertSheet(SHEET_NAME);

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Livreur','Téléphone','Ville','Montant','Statut','Date','Heure']);
    sheet.getRange(1,1,1,7).setFontWeight('bold');
  }

  // Lit les données (form-encoded via URLSearchParams)
  var data = e.parameter;
  const now = new Date();
  sheet.appendRow([
    data.livreur   || '',
    data.telephone || '',
    data.ville     || '',
    data.montant   || 0,
    'Livré',
    Utilities.formatDate(now, 'Africa/Porto-Novo', 'dd/MM/yyyy'),
    Utilities.formatDate(now, 'Africa/Porto-Novo', 'HH:mm')
  ]);

  return ContentService
    .createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME);
  if (!sheet) return json([]);

  const rows = sheet.getDataRange().getValues();
  const headers = rows[0];
  const data = rows.slice(1).map(r => {
    const obj = {};
    headers.forEach((h,i) => obj[h] = r[i]);
    return obj;
  }).reverse();

  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function json(d) {
  return ContentService
    .createTextOutput(JSON.stringify(d))
    .setMimeType(ContentService.MimeType.JSON);
}
